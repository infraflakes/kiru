fn run(rec project, txt message) {
  command("echo " + message)
    .in(project.dir)
    .direnv()
    .env({ KIRU_MESSAGE = message })
    .shell("/bin/sh")
    .timeout("5")
    .stream()
    .code();
  return(::helper() + project.name);
};
